# 构建你自己的编程语言

> 从零开始，用 Python 构建一个完整的编程语言，包含词法分析、语法分析、解释执行。

## 📋 项目简介

编程语言是程序员与计算机沟通的桥梁。通过从零构建一个编程语言，你将深入理解编译原理、抽象语法树、求值策略、类型系统等核心概念。

本教程将引导你构建 `MiniLang`——一个支持以下特性的编程语言：
- 变量绑定和函数定义
- 算术和逻辑运算
- 条件分支和循环
- 闭包和高阶函数
- 列表操作
- 交互式 REPL

## 🛠️ 技术栈选择

| 组件 | 技术 | 说明 |
|------|------|------|
| 语言 | Python 3.10+ | 实现语言 |

## 📖 核心概念解释

### 1. 编译/解释流程

```
源代码字符串
    │
    ▼
┌───────────────┐
│ 词法分析器     │ → Token 流
│ (Lexer)       │
└───────────────┘
    │
    ▼
┌───────────────┐
│ 语法分析器     │ → 抽象语法树 (AST)
│ (Parser)      │
└───────────────┘
    │
    ▼
┌───────────────┐
│ 语义分析       │ → 类型检查、作用域分析
│ (Analyzer)    │
└───────────────┘
    │
    ├──── 解释执行 ──→ 树遍历 / 字节码虚拟机
    │
    └──── 编译执行 ──→ 生成机器码 / LLVM IR
```

### 2. MiniLang 语法设计

```scheme
; 变量定义
(let x 10)
(let name "MiniLang")

; 函数定义
(define square (x) (* x x))
(define add (a b) (+ a b))

; 条件
(if (> x 5) "大" "小")

; Lambda 表达式
(let double (lambda (x) (* x 2)))

; 高阶函数
(map double (list 1 2 3 4 5))

; 递归
(define factorial (n)
  (if (<= n 1)
    1
    (* n (factorial (- n 1)))))

; 列表操作
(let nums (list 1 2 3 4 5))
(head nums)     ; → 1
(tail nums)     ; → (2 3 4 5)
(cons 0 nums)   ; → (0 1 2 3 4 5)
```

### 3. 抽象语法树 (AST)

```python
# 每种语言结构对应一个 AST 节点

# (let x 10) 的 AST:
LetNode(name='x', value=IntLiteral(10))

# (+ x (* y z)) 的 AST:
BinaryOpNode(
    op='+',
    left=VarNode('x'),
    right=BinaryOpNode(
        op='*',
        left=VarNode('y'),
        right=VarNode('z')
    )
)

# (define add (a b) (+ a b)) 的 AST:
DefineNode(
    name='add',
    params=['a', 'b'],
    body=BinaryOpNode(op='+', left=VarNode('a'), right=VarNode('b'))
)
```

### 4. 环境和作用域

```
全局环境 (Global)
├── +: <builtin>
├── -: <builtin>
├── *: <builtin>
└── ...

函数环境 (Factorial)
├── parent → Global
├── n: 5

```

### 第 1 步：词法分析器（Lexer）

```python
# lexer.py
from enum import Enum, auto
from dataclasses import dataclass
from typing import List

class TokenType(Enum):
    # 字面量
    INTEGER = auto()
    FLOAT = auto()
    STRING = auto()
    BOOLEAN = auto()
    NIL = auto()

    # 标识符
    IDENTIFIER = auto()

    # 关键字
    LET = auto()
    DEFINE = auto()
    IF = auto()
    LAMBDA = auto()
    WHILE = auto()
    DO = auto()
    BEGIN = auto()
    COND = auto()
    AND = auto()
    OR = auto()
    NOT = auto()

    # 分隔符
    LPAREN = auto()    # (
    RPAREN = auto()    # )
    LBRACKET = auto()  # [
    RBRACKET = auto()  # ]
    QUOTE = auto()     # '

    EOF = auto()

KEYWORDS = {
    'let': TokenType.LET, 'define': TokenType.DEFINE,
    'if': TokenType.IF, 'lambda': TokenType.LAMBDA,
    'while': TokenType.WHILE, 'do': TokenType.DO,
    'begin': TokenType.BEGIN, 'cond': TokenType.COND,
    'and': TokenType.AND, 'or': TokenType.OR,
    'not': TokenType.NOT,
    'true': TokenType.BOOLEAN, 'false': TokenType.BOOLEAN,
    'nil': TokenType.NIL,
}

@dataclass
class Token:
    type: TokenType
    value: object
    line: int
    column: int

    def __repr__(self):
        return f"Token({self.type.name}, {self.value!r})"

class Lexer:
    """词法分析器 - 将源代码转换为 Token 序列"""

    def __init__(self, source: str):
        self.source = source
        self.pos = 0
        self.line = 1
        self.column = 1
        self.tokens: List[Token] = []

    def tokenize(self) -> List[Token]:
        while self.pos < len(self.source):
            self._skip_whitespace_and_comments()

            if self.pos >= len(self.source):
                break

            ch = self.source[self.pos]

            if ch == '(':
                self._add_token(TokenType.LPAREN, '(')
            elif ch == ')':
                self._add_token(TokenType.RPAREN, ')')
            elif ch == '[':
                self._add_token(TokenType.LBRACKET, '[')
            elif ch == ']':
                self._add_token(TokenType.RBRACKET, ']')
            elif ch == "'":
                self._add_token(TokenType.QUOTE, "'")
            elif ch == '"':
                self._read_string()
            elif ch.isdigit() or (ch == '-' and self._peek_next_is_digit()):
                self._read_number()
            elif ch.isalpha() or ch in '_-+*/=<>!?':
                self._read_identifier()
            else:
                raise SyntaxError(f"无法识别的字符 '{ch}' 在第 {self.line} 行")

        self.tokens.append(Token(TokenType.EOF, None, self.line, self.column))
        return self.tokens

    def _skip_whitespace_and_comments(self):
        while self.pos < len(self.source):
            ch = self.source[self.pos]
            if ch in ' \t\r':
                self.pos += 1
                self.column += 1
            elif ch == '\n':
                self.pos += 1
                self.line += 1
                self.column = 1
            elif ch == ';':  # 注释
                while self.pos < len(self.source) and self.source[self.pos] != '\n':
                    self.pos += 1
            else:
                break

    def _read_string(self):
        self.pos += 1  # 跳过开始的 "
        start = self.pos
        value = []

        while self.pos < len(self.source) and self.source[self.pos] != '"':
            if self.source[self.pos] == '\\':
                self.pos += 1
                if self.pos < len(self.source):
                    escape = self.source[self.pos]
                    escapes = {'n': '\n', 't': '\t', 'r': '\r', '\\': '\\', '"': '"'}
                    value.append(escapes.get(escape, escape))
            else:
                value.append(self.source[self.pos])
            self.pos += 1

        if self.pos >= len(self.source):
            raise SyntaxError("未关闭的字符串")

        self.pos += 1  # 跳过结束的 "
        self.tokens.append(Token(TokenType.STRING, ''.join(value), self.line, self.column))

    def _read_number(self):
        start = self.pos
        has_dot = False
        is_negative = self.source[self.pos] == '-'
        if is_negative:
            self.pos += 1

        while self.pos < len(self.source):
            ch = self.source[self.pos]
            if ch.isdigit():
                self.pos += 1
            elif ch == '.' and not has_dot:
                has_dot = True
                self.pos += 1
            else:
                break

        value_str = self.source[start:self.pos]
        if has_dot:
            self.tokens.append(Token(TokenType.FLOAT, float(value_str), self.line, self.column))
        else:
            self.tokens.append(Token(TokenType.INTEGER, int(value_str), self.line, self.column))

    def _read_identifier(self):
        start = self.pos
        while self.pos < len(self.source) and self.source[self.pos] not in ' \t\n\r()[]\'\";':
            self.pos += 1

        value = self.source[start:self.pos]

        if value in KEYWORDS:
            token_type = KEYWORDS[value]
            token_value = value == 'true' if token_type == TokenType.BOOLEAN else value
            self.tokens.append(Token(token_type, token_value, self.line, self.column))
        else:
            self.tokens.append(Token(TokenType.IDENTIFIER, value, self.line, self.column))

    def _add_token(self, token_type: TokenType, value: object):
        self.tokens.append(Token(token_type, value, self.line, self.column))
        self.pos += 1
        self.column += 1

    def _peek_next_is_digit(self) -> bool:
        if self.pos + 1 < len(self.source):
            return self.source[self.pos + 1].isdigit()
        return False
```

### 第 3 步：递归下降解析器

```python
# parser.py
from typing import List
from lexer import Token, TokenType
from ast_nodes import *

class Parser:
    """递归下降解析器 - 将 Token 流转换为 AST"""

    def __init__(self, tokens: List[Token]):
        self.tokens = tokens
        self.pos = 0

    def parse(self) -> List[ASTNode]:
        """解析整个程序，返回顶层表达式列表"""
        expressions = []
        while self._current_type() != TokenType.EOF:
            expressions.append(self._parse_expr())
        return expressions

    def _current(self) -> Token:
        return self.tokens[self.pos]

    def _current_type(self) -> TokenType:
        return self.tokens[self.pos].type

    def _advance(self) -> Token:
        token = self.tokens[self.pos]
        self.pos += 1
        return token

    def _expect(self, token_type: TokenType) -> Token:
        if self._current_type() != token_type:
            raise SyntaxError(
                f"期望 {token_type.name}，"
                f"但得到 {self._current_type().name} "
                f"'{self._current().value}' 在第 {self._current().line} 行"
            )
        return self._advance()

    def _parse_expr(self) -> ASTNode:
        """解析一个表达式"""
        token_type = self._current_type()

        if token_type == TokenType.LPAREN:
            return self._parse_list_expr()
        elif token_type == TokenType.INTEGER:
            return IntLiteral(self._advance().value)
        elif token_type == TokenType.FLOAT:
            return FloatLiteral(self._advance().value)
        elif token_type == TokenType.STRING:
            return StringLiteral(self._advance().value)
        elif token_type == TokenType.BOOLEAN:
            return BoolLiteral(self._advance().value)
        elif token_type == TokenType.NIL:
            self._advance()
            return NilLiteral()
        elif token_type == TokenType.IDENTIFIER:
            return VarRef(self._advance().value)
        else:
            raise SyntaxError(
                f"无法解析的 Token: {self._current()} "
                f"在第 {self._current().line} 行"
            )

    def _parse_list_expr(self) -> ASTNode:
        """解析括号包裹的表达式"""
        self._expect(TokenType.LPAREN)
        keyword = self._current_type()

        if keyword == TokenType.LET:
            return self._parse_let()
        elif keyword == TokenType.DEFINE:
            return self._parse_define()
        elif keyword == TokenType.LAMBDA:
            return self._parse_lambda()
        elif keyword == TokenType.IF:
            return self._parse_if()
        elif keyword == TokenType.WHILE:
            return self._parse_while()
        elif keyword == TokenType.BEGIN:
            return self._parse_begin()
        elif keyword == TokenType.COND:
            return self._parse_cond()
        else:
            return self._parse_call()

    def _parse_let(self) -> LetExpr:
        self._advance()  # 消费 'let'
        name = self._expect(TokenType.IDENTIFIER).value
        value = self._parse_expr()
        self._expect(TokenType.RPAREN)
        return LetExpr(name, value)

    def _parse_define(self) -> DefineExpr:
        self._advance()  # 消费 'define'
        name = self._expect(TokenType.IDENTIFIER).value
        self._expect(TokenType.LPAREN)

        params = []
        while self._current_type() != TokenType.RPAREN:
            params.append(self._expect(TokenType.IDENTIFIER).value)
        self._advance()  # 消费 ')'

        body = self._parse_expr()
        self._expect(TokenType.RPAREN)
        return DefineExpr(name, params, body)

    def _parse_lambda(self) -> LambdaExpr:
        self._advance()  # 消费 'lambda'
        self._expect(TokenType.LPAREN)

        params = []
        while self._current_type() != TokenType.RPAREN:
            params.append(self._expect(TokenType.IDENTIFIER).value)
        self._advance()

        body = self._parse_expr()
        self._expect(TokenType.RPAREN)
        return LambdaExpr(params, body)

    def _parse_if(self) -> IfExpr:
        self._advance()  # 消费 'if'
        condition = self._parse_expr()
        then_branch = self._parse_expr()

        else_branch = None
        if self._current_type() != TokenType.RPAREN:
            else_branch = self._parse_expr()

        self._expect(TokenType.RPAREN)
        return IfExpr(condition, then_branch, else_branch)

    def _parse_while(self) -> WhileExpr:
        self._advance()
        condition = self._parse_expr()
        body = []
        while self._current_type() != TokenType.RPAREN:
            body.append(self._parse_expr())
        self._expect(TokenType.RPAREN)
        return WhileExpr(condition, body)

    def _parse_begin(self) -> BeginExpr:
        self._advance()
        body = []
        while self._current_type() != TokenType.RPAREN:
            body.append(self._parse_expr())
        self._expect(TokenType.RPAREN)
        return BeginExpr(body)

    def _parse_cond(self) -> CondExpr:
        self._advance()
        branches = []
        while self._current_type() == TokenType.LPAREN:
            self._advance()
            condition = self._parse_expr()
            body = self._parse_expr()
            self._expect(TokenType.RPAREN)
            branches.append((condition, body))
        self._expect(TokenType.RPAREN)
        return CondExpr(branches)

    def _parse_call(self) -> CallExpr:
        function = self._parse_expr()
        arguments = []
        while self._current_type() != TokenType.RPAREN:
            arguments.append(self._parse_expr())
        self._expect(TokenType.RPAREN)
        return CallExpr(function, arguments)
```

### 第 4 步：求值器（解释器）

```python
# evaluator.py
from ast_nodes import *
from typing import Any, Callable
import operator

class Environment:
    """作用域环境 - 管理变量绑定"""

    def __init__(self, parent: 'Environment' = None):
        self.bindings: dict[str, Any] = {}
        self.parent = parent

    def get(self, name: str) -> Any:
        if name in self.bindings:
            return self.bindings[name]
        if self.parent:
            return self.parent.get(name)
        raise NameError(f"未定义的变量: '{name}'")

    def set(self, name: str, value: Any):
        self.bindings[name] = value

    def extend(self) -> 'Environment':
        """创建一个新的子作用域"""
        return Environment(parent=self)


class Function:
    """用户定义函数（支持闭包）"""

    def __init__(self, params: List[str], body: ASTNode, closure: Environment):
        self.params = params
        self.body = body
        self.closure = closure  # 闭包：捕获定义时的环境

    def __repr__(self):
        return f"<function({', '.join(self.params)})>"


class Evaluator:
    """树遍历求值器"""

    def __init__(self):
        self.global_env = Environment()
        self._setup_builtins()

    def _setup_builtins(self):
        """注册内置函数"""
        builtins = {
            '+': lambda args: sum(args),
            '-': lambda args: args[0] - sum(args[1:]) if len(args) > 1 else -args[0],
            '*': lambda args: self._product(args),
            '/': lambda args: args[0] / args[1],
            '%': lambda args: args[0] % args[1],
            '=': lambda args: args[0] == args[1],
            '<>': lambda args: args[0] != args[1],
            '<': lambda args: args[0] < args[1],
            '>': lambda args: args[0] > args[1],
            '<=': lambda args: args[0] <= args[1],
            '>=': lambda args: args[0] >= args[1],
            'and': lambda args: all(args),
            'or': lambda args: any(args),
            'not': lambda args: not args[0],
            'print': lambda args: self._builtin_print(args),
            'list': lambda args: list(args),
            'head': lambda args: args[0][0] if args[0] else None,
            'tail': lambda args: args[0][1:] if len(args[0]) > 1 else [],
            'cons': lambda args: [args[0]] + (args[1] if isinstance(args[1], list) else [args[1]]),
            'len': lambda args: len(args[0]),
            'empty?': lambda args: len(args[0]) == 0,
            'map': lambda args: self._builtin_map(args),
            'filter': lambda args: self._builtin_filter(args),
            'reduce': lambda args: self._builtin_reduce(args),
            'range': lambda args: list(range(*args)),
            'to-string': lambda args: str(args[0]),
            'to-int': lambda args: int(args[0]),
            'type': lambda args: type(args[0]).__name__,
        }

        for name, func in builtins.items():
            self.global_env.set(name, func)

    def evaluate(self, node: ASTNode, env: Environment = None) -> Any:
        """求值 AST 节点"""
        if env is None:
            env = self.global_env

        if isinstance(node, IntLiteral):
            return node.value
        elif isinstance(node, FloatLiteral):
            return node.value
        elif isinstance(node, StringLiteral):
            return node.value
        elif isinstance(node, BoolLiteral):
            return node.value
        elif isinstance(node, NilLiteral):
            return None
        elif isinstance(node, VarRef):
            return env.get(node.name)
        elif isinstance(node, LetExpr):
            return self._eval_let(node, env)
        elif isinstance(node, DefineExpr):
            return self._eval_define(node, env)
        elif isinstance(node, LambdaExpr):
            return Function(node.params, node.body, env)
        elif isinstance(node, IfExpr):
            return self._eval_if(node, env)
        elif isinstance(node, WhileExpr):
            return self._eval_while(node, env)
        elif isinstance(node, BeginExpr):
            return self._eval_begin(node, env)
        elif isinstance(node, CondExpr):
            return self._eval_cond(node, env)
        elif isinstance(node, CallExpr):
            return self._eval_call(node, env)
        else:
            raise RuntimeError(f"未知的 AST 节点: {type(node).__name__}")

    def _eval_let(self, node: LetExpr, env: Environment) -> Any:
        value = self.evaluate(node.value, env)
        env.set(node.name, value)
        return value

    def _eval_define(self, node: DefineExpr, env: Environment) -> Any:
        func = Function(node.params, node.body, env)
        env.set(node.name, func)
        return func

    def _eval_if(self, node: IfExpr, env: Environment) -> Any:
        condition = self.evaluate(node.condition, env)
        if condition and condition != 0 and condition != []:
            return self.evaluate(node.then_branch, env)
        elif node.else_branch:
            return self.evaluate(node.else_branch, env)
        return None

    def _eval_while(self, node: WhileExpr, env: Environment) -> Any:
        result = None
        while self.evaluate(node.condition, env):
            for expr in node.body:
                result = self.evaluate(expr, env)
        return result

    def _eval_begin(self, node: BeginExpr, env: Environment) -> Any:
        result = None
        for expr in node.body:
            result = self.evaluate(expr, env)
        return result

    def _eval_cond(self, node: CondExpr, env: Environment) -> Any:
        for condition, body in node.branches:
            if self.evaluate(condition, env):
                return self.evaluate(body, env)
        return None

    def _eval_call(self, node: CallExpr, env: Environment) -> Any:
        function = self.evaluate(node.function, env)
        args = [self.evaluate(arg, env) for arg in node.arguments]

        if callable(function) and not isinstance(function, Function):
            # 内置函数
            return function(args)
        elif isinstance(function, Function):
            # 用户定义函数 - 创建新作用域
            func_env = function.closure.extend()
            for param, arg in zip(function.params, args):
                func_env.set(param, arg)
            return self.evaluate(function.body, func_env)
        else:
            raise RuntimeError(f"不可调用: {function}")

    def _product(self, args):
        result = 1
        for a in args:
            result *= a
        return result

    def _builtin_print(self, args):
        output = ' '.join(str(a) for a in args)
        print(output)
        return args[0] if len(args) == 1 else args

    def _builtin_map(self, args):
        func, lst = args[0], args[1]
        return [self._apply_func(func, [x]) for x in lst]

    def _builtin_filter(self, args):
        func, lst = args[0], args[1]
        return [x for x in lst if self._apply_func(func, [x])]

    def _builtin_reduce(self, args):
        func, lst, init = args[0], args[1], args[2]
        result = init
        for x in lst:
            result = self._apply_func(func, [result, x])
        return result

    def _apply_func(self, func, args):
        if callable(func) and not isinstance(func, Function):
            return func(args)
        elif isinstance(func, Function):
            env = func.closure.extend()
            for param, arg in zip(func.params, args):
                env.set(param, arg)
            return self.evaluate(func.body, env)
```

### 第 5 步：REPL 交互式环境

```python
# repl.py
"""MiniLang REPL - 交互式编程环境"""
from lexer import Lexer
from parser import Parser
from evaluator import Evaluator

def run(source: str, evaluator: Evaluator) -> any:
    """运行源代码"""
    tokens = Lexer(source).tokenize()
    ast = Parser(tokens).parse()
    result = None
    for node in ast:
        result = evaluator.evaluate(node)
    return result

def repl():
    """启动 REPL"""
    print("╔══════════════════════════════════════════╗")
    print("║   MiniLang v0.1 - 从零构建的编程语言     ║")
    print("║   输入 (quit) 退出，(help) 帮助          ║")
    print("╚══════════════════════════════════════════╝")

    evaluator = Evaluator()

    while True:
        try:
            source = input("\nminilang> ").strip()
            if not source:
                continue
            if source in ('quit', 'exit', 'q'):
                break
            if source == 'help':
                print(HELP_TEXT)
                continue

            result = run(source, evaluator)
            if result is not None:
                print(f"=> {result}")

        except Exception as e:
            print(f"错误: {e}")

    print("再见！")

HELP_TEXT = """
MiniLang 语法:
  (let x 10)                    ; 定义变量
  (define f (x) (* x x))        ; 定义函数
  (if (> x 5) "大" "小")        ; 条件表达式
  (lambda (x) (* x 2))          ; Lambda 表达式
  (while (< x 10) (let x (+ x 1)))  ; 循环
  (begin (print 1) (print 2))   ; 顺序执行

内置函数:
  + - * / % = < > <= >= and or not
  list head tail cons len empty?
  map filter reduce range
  print to-string to-int type
"""

if __name__ == "__main__":
    repl()
```
