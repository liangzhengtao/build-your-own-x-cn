# 构建你自己的区块链

> 从零开始，用 Python 构建一个支持工作量证明的区块链。

## 📋 项目简介

区块链是一种去中心化的分布式账本技术，支撑着比特币、以太坊等加密货币。通过从零构建一个区块链，你将深入理解哈希链、共识机制、P2P 网络、数字签名等核心概念。

本教程将引导你构建 `MiniChain`，支持：
- 工作量证明（PoW）挖矿
- 交易创建与验证
- 区块链有效性验证
- 简单的 P2P 网络
- REST API 接口
- 钱包与数字签名

## 📖 核心概念解释

### 1. 区块链结构

```
┌─────────────────────────────────────────────────────┐
│ 区块 #0 (创世区块)                                    │
│ ┌─────────────────────────────────────────────────┐ │
│ │ 前一区块哈希: 0000...0000                        │ │
│ │ 时间戳: 2024-01-01 00:00:00                     │ │
│ │ Nonce: 0                                        │ │
│ │ Merkle 根: a1b2c3...                            │ │
│ │ 交易: [coinbase tx]                             │ │
│ └─────────────────────────────────────────────────┘ │
│ 区块哈希: 0000abc123...                              │
└─────────────────────────────────────────────────────┘
                         │
                         ▼
┌─────────────────────────────────────────────────────┐
│ 区块 #1                                              │
│ ┌─────────────────────────────────────────────────┐ │
│ │ 前一区块哈希: 0000abc123... ← 指向区块 #0        │ │
│ │ 时间戳: 2024-01-01 00:10:00                     │ │
│ │ Nonce: 45283                                    │ │
│ │ 交易: [coinbase tx, Alice→Bob 5 BTC]            │ │
│ └─────────────────────────────────────────────────┘ │
│ 区块哈希: 0000def456...                              │
└─────────────────────────────────────────────────────┘
                         │
                         ▼
┌─────────────────────────────────────────────────────┐
│ 区块 #2                                              │
│ ┌─────────────────────────────────────────────────┐ │
│ │ 前一区块哈希: 0000def456... ← 指向区块 #1        │ │
│ │ ...                                              │ │
│ └─────────────────────────────────────────────────┘ │
└─────────────────────────────────────────────────────┘
```

### 2. 工作量证明（Proof of Work）

矿工需要找到一个 nonce，使得区块哈希满足难度要求：

```
找到 nonce 使得:
  SHA256(block_header + nonce) < target

示例（难度 4）:
  target = 0x0000FFFFFFFFFFFFFFFF...
  需要哈希值以 4 个零开头

  nonce=0     → hash=8b7f3a...  ✗
  nonce=1     → hash=2c1d5e...  ✗
  ...
  nonce=45283 → hash=0000a3...  ✓ 找到了！
```

### 3. 交易与 UTXO

```
简化模型（账户余额模型）:
  Alice: 100 BTC
  Alice → Bob: 30 BTC
  Alice: 70 BTC, Bob: 30 BTC

UTXO 模型（比特币使用）:
  UTXO #1: Alice 拥有 100 BTC

  交易: 输入 UTXO #1 → 输出:
    - 30 BTC → Bob 的地址
    - 69.99 → Alice 的找零地址
    - 0.01 → 矿工手续费

  UTXO #1 被消费，产生两个新的 UTXO
```

### 4. Merkle 树

```
        Merkle Root
       /            \
   Hash(AB)       Hash(CD)
    /    \         /    \
 Hash(A) Hash(B) Hash(C) Hash(D)
   |       |       |       |
  Tx A    Tx B    Tx C    Tx D
```

## 🚀 分步实现指南

### 第 1 步：区块结构

```python
# blockchain/block.py
import hashlib
import json
import time
from typing import List

class Transaction:
    """交易"""

    def __init__(self, sender: str, recipient: str, amount: float,
                 fee: float = 0.0, signature: str = None):
        self.sender = sender
        self.recipient = recipient
        self.amount = amount
        self.fee = fee
        self.signature = signature
        self.timestamp = time.time()

    def to_dict(self) -> dict:
        return {
            'sender': self.sender,
            'recipient': self.recipient,
            'amount': self.amount,
            'fee': self.fee,
            'timestamp': self.timestamp,
        }

    def compute_hash(self) -> str:
        tx_string = json.dumps(self.to_dict(), sort_keys=True)
        return hashlib.sha256(tx_string.encode()).hexdigest()

    def is_coinbase(self) -> bool:
        return self.sender == "COINBASE"


class Block:
    """区块"""

    def __init__(self, index: int, transactions: List[Transaction],
                 previous_hash: str, nonce: int = 0):
        self.index = index
        self.timestamp = time.time()
        self.transactions = transactions
        self.previous_hash = previous_hash
        self.nonce = nonce
        self.merkle_root = self._compute_merkle_root()

    def compute_hash(self) -> str:
        block_data = {
            'index': self.index,
            'timestamp': self.timestamp,
            'transactions': [tx.to_dict() for tx in self.transactions],
            'previous_hash': self.previous_hash,
            'nonce': self.nonce,
            'merkle_root': self.merkle_root,
        }
        block_string = json.dumps(block_data, sort_keys=True)
        return hashlib.sha256(block_string.encode()).hexdigest()

    def _compute_merkle_root(self) -> str:
        """计算 Merkle 根"""
        if not self.transactions:
            return hashlib.sha256(b'').hexdigest()

        tx_hashes = [tx.compute_hash() for tx in self.transactions]

        while len(tx_hashes) > 1:
            if len(tx_hashes) % 2 != 0:
                tx_hashes.append(tx_hashes[-1])  # 复制最后一个

            next_level = []
            for i in range(0, len(tx_hashes), 2):
                combined = tx_hashes[i] + tx_hashes[i + 1]
                next_level.append(hashlib.sha256(combined.encode()).hexdigest())
            tx_hashes = next_level

        return tx_hashes[0]

    def to_dict(self) -> dict:
        return {
            'index': self.index,
            'timestamp': self.timestamp,
            'transactions': [tx.to_dict() for tx in self.transactions],
            'previous_hash': self.previous_hash,
            'nonce': self.nonce,
            'merkle_root': self.merkle_root,
            'hash': self.compute_hash(),
        }
```

### 第 2 步：区块链核心

```python
# blockchain/chain.py
import time
from typing import List, Optional
from .block import Block, Transaction

class Blockchain:
    """区块链核心"""

    MINING_DIFFICULTY = 4       # 难度（哈希前缀零的个数）
    MINING_REWARD = 50.0        # 挖矿奖励
    BLOCK_SIZE_LIMIT = 10       # 每区块最大交易数
    BLOCK_TIME = 10.0           # 目标出块时间（秒）
    HALVING_INTERVAL = 210000   # 奖励减半周期

    def __init__(self):
        self.chain: List[Block] = []
        self.pending_transactions: List[Transaction] = []
        self.balances: dict[str, float] = {}
        self.difficulty = self.MINING_DIFFICULTY

        # 创建创世区块
        self._create_genesis_block()

    def _create_genesis_block(self):
        """创建创世区块"""
        genesis_tx = Transaction("COINBASE", "genesis", 0)
        genesis_block = Block(0, [genesis_tx], "0" * 64)
        self.chain.append(genesis_block)

    @property
    def last_block(self) -> Block:
        return self.chain[-1]

    def add_transaction(self, transaction: Transaction) -> bool:
        """添加交易到待处理池"""
        # 验证交易
        if transaction.is_coinbase():
            self.pending_transactions.append(transaction)
            return True

        # 检查余额
        sender_balance = self.balances.get(transaction.sender, 0)
        if sender_balance < transaction.amount + transaction.fee:
            raise ValueError(f"余额不足: {transaction.sender} "
                           f"有 {sender_balance}，需要 {transaction.amount + transaction.fee}")

        self.pending_transactions.append(transaction)
        return True

    def mine_block(self, miner_address: str) -> Optional[Block]:
        """挖矿 - 创建新区块"""
        if not self.pending_transactions:
            return None

        # 创建 coinbase 交易（挖矿奖励）
        coinbase = Transaction("COINBASE", miner_address, self.MINING_REWARD)

        # 选择交易
        selected_txs = [coinbase] + self.pending_transactions[:self.BLOCK_SIZE_LIMIT]

        # 创建新区块
        new_block = Block(
            index=len(self.chain),
            transactions=selected_txs,
            previous_hash=self.last_block.compute_hash()
        )

        # 工作量证明
        new_block = self._proof_of_work(new_block)

        # 更新余额
        self._update_balances(selected_txs)

        # 添加到链
        self.chain.append(new_block)

        # 清理已处理的交易
        self.pending_transactions = self.pending_transactions[self.BLOCK_SIZE_LIMIT:]

        return new_block

    def _proof_of_work(self, block: Block) -> Block:
        """工作量证明 - 寻找满足难度要求的 nonce"""
        target = "0" * self.difficulty

        while True:
            block.nonce += 1
            hash_result = block.compute_hash()

            if hash_result.startswith(target):
                print(f"找到 nonce: {block.nonce}, hash: {hash_result}")
                return block

    def _update_balances(self, transactions: List[Transaction]):
        """更新账户余额"""
        for tx in transactions:
            if tx.is_coinbase():
                self.balances[tx.recipient] = self.balances.get(tx.recipient, 0) + tx.amount
            else:
                self.balances[tx.sender] = self.balances.get(tx.sender, 0) - tx.amount - tx.fee
                self.balances[tx.recipient] = self.balances.get(tx.recipient, 0) + tx.amount
                if tx.fee > 0:
                    # 手续费给矿工（简化处理）
                    pass

    def is_chain_valid(self) -> bool:
        """验证整条链的有效性"""
        for i in range(1, len(self.chain)):
            current = self.chain[i]
            previous = self.chain[i - 1]

            # 验证哈希
            if current.previous_hash != previous.compute_hash():
                print(f"区块 {i} 的前哈希不匹配")
                return False

            # 验证工作量证明
            if not current.compute_hash().startswith("0" * self.difficulty):
                print(f"区块 {i} 的哈希不满足难度要求")
                return False

            # 验证 Merkle 根
            if current.merkle_root != current._compute_merkle_root():
                print(f"区块 {i} 的 Merkle 根不匹配")
                return False

        return True

    def get_balance(self, address: str) -> float:
        return self.balances.get(address, 0.0)

    def to_dict(self) -> dict:
        return {
            'chain': [block.to_dict() for block in self.chain],
            'length': len(self.chain),
            'difficulty': self.difficulty,
        }
```

### 第 3 步：钱包与数字签名

```python
# wallet/wallet.py
import hashlib
import json
from ecdsa import SigningKey, VerifyingKey, SECP256k1

class Wallet:
    """区块链钱包"""

    def __init__(self):
        self.private_key = SigningKey.generate(curve=SECP256k1)
        self.public_key = self.private_key.get_verifying_key()

    @property
    def address(self) -> str:
        """从公钥生成地址"""
        pub_key_bytes = self.public_key.to_string()
        # 简化版：直接对公钥哈希
        sha256_hash = hashlib.sha256(pub_key_bytes).hexdigest()
        return f"0x{sha256_hash[:40]}"

    def sign_transaction(self, transaction) -> str:
        """签名交易"""
        tx_data = json.dumps(transaction.to_dict(), sort_keys=True)
        signature = self.private_key.sign(tx_data.encode())
        return signature.hex()

    def verify_signature(self, transaction, signature_hex: str,
                        public_key_bytes: bytes) -> bool:
        """验证签名"""
        try:
            tx_data = json.dumps(transaction.to_dict(), sort_keys=True)
            vk = VerifyingKey.from_string(public_key_bytes, curve=SECP256k1)
            return vk.verify(bytes.fromhex(signature_hex), tx_data.encode())
        except Exception:
            return False
```

### 第 4 步：REST API 服务

```python
# api/server.py
from flask import Flask, jsonify, request
from blockchain.chain import Blockchain
from blockchain.block import Transaction
from wallet.wallet import Wallet

app = Flask(__name__)
blockchain = Blockchain()
wallets: dict[str, Wallet] = {}

@app.route('/chain', methods=['GET'])
def get_chain():
    """获取完整区块链"""
    return jsonify(blockchain.to_dict()), 200

@app.route('/mine', methods=['POST'])
def mine():
    """挖矿"""
    data = request.get_json()
    miner_address = data.get('miner_address', 'default_miner')

    block = blockchain.mine_block(miner_address)

    if block:
        return jsonify({
            'message': '新区块已挖出',
            'block': block.to_dict(),
        }), 200
    else:
        return jsonify({'message': '没有待处理的交易'}), 200

@app.route('/transaction/new', methods=['POST'])
def new_transaction():
    """创建新交易"""
    data = request.get_json()

    required = ['sender', 'recipient', 'amount']
    if not all(k in data for k in required):
        return jsonify({'error': '缺少必要字段'}), 400

    try:
        tx = Transaction(
            sender=data['sender'],
            recipient=data['recipient'],
            amount=float(data['amount']),
            fee=float(data.get('fee', 0)),
        )
        blockchain.add_transaction(tx)
        return jsonify({
            'message': '交易已添加到待处理池',
            'transaction_hash': tx.compute_hash(),
        }), 201
    except ValueError as e:
        return jsonify({'error': str(e)}), 400

@app.route('/balance/<address>', methods=['GET'])
def get_balance(address):
    """查询余额"""
    balance = blockchain.get_balance(address)
    return jsonify({'address': address, 'balance': balance}), 200

@app.route('/validate', methods=['GET'])
def validate_chain():
    """验证区块链"""
    is_valid = blockchain.is_chain_valid()
    return jsonify({
        'valid': is_valid,
        'chain_length': len(blockchain.chain),
    }), 200

@app.route('/wallet/new', methods=['POST'])
def create_wallet():
    """创建新钱包"""
    wallet = Wallet()
    wallets[wallet.address] = wallet
    return jsonify({
        'address': wallet.address,
        'private_key': wallet.private_key.to_string().hex(),
        'public_key': wallet.public_key.to_string().hex(),
    }), 201

if __name__ == '__main__':
    app.run(port=5000, debug=True)
```

### 第 5 步：命令行挖矿

```python
# main.py
"""MiniChain - 从零构建的区块链"""
from blockchain.chain import Blockchain
from blockchain.block import Transaction
from wallet.wallet import Wallet
import time

def main():
    print("╔══════════════════════════════════════╗")
    print("║   MiniChain v0.1 - 区块链            ║")
    print("╚══════════════════════════════════════╝")

    chain = Blockchain()

    # 创建钱包
    alice = Wallet()
    bob = Wallet()
    miner = Wallet()

    print(f"\nAlice 地址: {alice.address}")
    print(f"Bob 地址:   {bob.address}")
    print(f"矿工 地址:  {miner.address}")

    # 挖第一个区块
    print("\n--- 开始挖矿 ---")
    tx1 = Transaction("COINBASE", miner.address, 50)
    chain.add_transaction(tx1)
    block = chain.mine_block(miner.address)
    if block:
        print(f"区块 #{block.index} 已挖出，nonce: {block.nonce}")

    # 进行一笔转账
    print("\n--- 转账: 矿工 → Alice 20 ---")
    tx2 = Transaction(miner.address, alice.address, 20, fee=0.1)
    chain.add_transaction(tx2)
    block = chain.mine_block(miner.address)
    if block:
        print(f"区块 #{block.index} 已挖出")

    # 查看余额
    print(f"\n矿工余额: {chain.get_balance(miner.address)}")
    print(f"Alice 余额: {chain.get_balance(alice.address)}")

    # 验证链
    print(f"\n链是否有效: {chain.is_chain_valid()}")
    print(f"链长度: {len(chain.chain)}")

if __name__ == "__main__":
    main()
```
