## 📖 核心概念解释

### 1. 搜索引擎架构

```
用户查询 "Python 教程"
    │
    ▼
┌───────────────┐     ┌───────────────┐     ┌───────────────┐
│  查询解析器    │────→│  倒排索引      │────→│  排序器       │
│  (Query       │     │  (Inverted    │     │  (Ranker)     │
│   Parser)     │     │   Index)      │     │  TF-IDF/BM25  │
└───────────────┘     └───────────────┘     └───────────────┘
                             ▲                      │
                             │                      ▼
                      ┌───────────────┐     ┌───────────────┐
                      │  网络爬虫      │     │  搜索结果     │
                      │  (Crawler)    │     │  (Results)    │
                      └───────────────┘     └───────────────┘
                             │
                             ▼
                        互联网网页
```

### 2. 倒排索引

倒排索引是搜索引擎的核心数据结构：

```
文档集合:
  Doc 1: "Python 是一种编程语言"
  Doc 2: "Python 教程很有趣"
  Doc 3: "编程语言有很多种"

倒排索引:
  "Python"    → [Doc 1, Doc 2]
  "编程"      → [Doc 1, Doc 3]
  "语言"      → [Doc 1, Doc 3]
  "教程"      → [Doc 2]
  "有趣"      → [Doc 2]
  "很多"      → [Doc 3]

搜索 "Python 编程":
  "Python"  ∩ "编程"  → Doc 1 (两个词都出现)
  "Python"  ∪ "编程"  → Doc 1, Doc 2, Doc 3 (任一出现)
```

### 3. TF-IDF 算法

```
TF (词频) = 词在文档中出现的次数 / 文档总词数
IDF (逆文档频率) = log(总文档数 / 包含该词的文档数)
TF-IDF = TF × IDF

示例:
  文档 1: "Python Python 教程"  (共 3 词)
  TF("Python", Doc1) = 2/3 ≈ 0.667

  总文档数 = 1000, 包含 "Python" 的文档 = 100
  IDF("Python") = log(1000/100) = log(10) ≈ 2.303

  TF-IDF("Python", Doc1) = 0.667 × 2.303 ≈ 1.535
```

### 4. 网络爬虫工作原理

```
种子 URL → 获取页面 → 解析链接 → 加入队列 → 重复
  │          │          │          │
  ▼          ▼          ▼          ▼
URL 队列   HTML 内容   新发现的    BFS/DFS
                     URL       遍历策略
```

## 🚀 分步实现指南

### 第 1 步：网页爬虫

```python
# crawler/spider.py
import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin, urlparse
from collections import deque
import time
import re

class WebPage:
    """网页数据"""
    def __init__(self, url: str, title: str, content: str, links: list):
        self.url = url
        self.title = title
        self.content = content
        self.links = links

class WebCrawler:
    """网络爬虫"""

    def __init__(self, max_pages: int = 100, delay: float = 1.0,
                 user_agent: str = "MiniSearch/1.0"):
        self.max_pages = max_pages
        self.delay = delay
        self.user_agent = user_agent
        self.visited: set[str] = set()
        self.queue: deque[str] = deque()
        self.pages: list[WebPage] = []

    def crawl(self, seed_urls: list[str]) -> list[WebPage]:
        """BFS 爬取网页"""
        for url in seed_urls:
            self.queue.append(url)

        while self.queue and len(self.visited) < self.max_pages:
            url = self.queue.popleft()

            if url in self.visited:
                continue

            try:
                page = self._fetch_page(url)
                if page:
                    self.pages.append(page)
                    print(f"[{len(self.visited)}/{self.max_pages}] {page.title[:50]}")

                    # 将新链接加入队列
                    for link in page.links:
                        if link not in self.visited:
                            self.queue.append(link)

                time.sleep(self.delay)  # 礼貌爬取

            except Exception as e:
                print(f"爬取失败 {url}: {e}")

        return self.pages

    def _fetch_page(self, url: str) -> WebPage | None:
        """获取并解析单个页面"""
        self.visited.add(url)

        headers = {'User-Agent': self.user_agent}
        response = requests.get(url, headers=headers, timeout=10)

        if response.status_code != 200:
            return None

        content_type = response.headers.get('Content-Type', '')
        if 'text/html' not in content_type:
            return None

        soup = BeautifulSoup(response.text, 'html.parser')

        # 提取标题
        title = soup.title.string if soup.title else url

        # 提取文本内容
        for tag in soup(['script', 'style', 'nav', 'footer', 'header']):
            tag.decompose()

        content = soup.get_text(separator=' ', strip=True)
        content = re.sub(r'\s+', ' ', content)  # 合并多余空白

        # 提取链接
        links = []
        for a_tag in soup.find_all('a', href=True):
            href = urljoin(url, a_tag['href'])
            parsed = urlparse(href)
            if parsed.scheme in ('http', 'https'):
                # 去除 fragment
                clean_url = f"{parsed.scheme}://{parsed.netloc}{parsed.path}"
                if parsed.query:
                    clean_url += f"?{parsed.query}"
                links.append(clean_url)

        return WebPage(url, title, content, links)
```

### 第 2 步：倒排索引

```python
# index/inverted_index.py
import math
import re
from collections import defaultdict
from dataclasses import dataclass, field
from typing import List, Dict, Set

@dataclass
class DocumentInfo:
    """文档元信息"""
    doc_id: int
    url: str
    title: str
    length: int  # 文档词数

@dataclass
class PostingEntry:
    """倒排列表条目"""
    doc_id: int
    term_frequency: int   # 词在文档中出现的次数
    positions: List[int]  # 词在文档中的位置

class InvertedIndex:
    """倒排索引"""

    def __init__(self):
        # 核心数据结构: term → [PostingEntry, ...]
        self.index: Dict[str, List[PostingEntry]] = defaultdict(list)
        self.documents: Dict[int, DocumentInfo] = {}
        self.doc_count = 0
        self.total_terms = 0

    def add_document(self, url: str, title: str, content: str) -> int:
        """添加文档到索引"""
        doc_id = self.doc_count
        self.doc_count += 1

        # 分词
        tokens = self._tokenize(title + " " + content)
        doc_length = len(tokens)

        self.documents[doc_id] = DocumentInfo(doc_id, url, title, doc_length)

        # 统计词频和位置
        term_positions: Dict[str, List[int]] = defaultdict(list)
        for position, token in enumerate(tokens):
            term_positions[token].append(position)

        # 构建倒排列表
        for term, positions in term_positions.items():
            entry = PostingEntry(
                doc_id=doc_id,
                term_frequency=len(positions),
                positions=positions,
            )
            self.index[term].append(entry)
            self.total_terms += 1

        return doc_id

    def search(self, query: str, top_k: int = 10) -> List[tuple]:
        """搜索并返回排序后的结果"""
        query_terms = self._tokenize(query)
        if not query_terms:
            return []

        # 收集包含查询词的所有文档
        candidate_docs: Dict[int, float] = defaultdict(float)

        for term in query_terms:
            if term not in self.index:
                continue

            idf = self._compute_idf(term)

            for posting in self.index[term]:
                tf = self._compute_tf(posting.term_frequency,
                                      self.documents[posting.doc_id].length)
                score = tf * idf
                candidate_docs[posting.doc_id] += score

        # 按分数排序
        sorted_docs = sorted(candidate_docs.items(),
                           key=lambda x: x[1], reverse=True)

        # 返回 top-k 结果
        results = []
        for doc_id, score in sorted_docs[:top_k]:
            doc = self.documents[doc_id]
            results.append((doc, score))

        return results

    def _compute_tf(self, term_freq: int, doc_length: int) -> float:
        """计算词频（归一化）"""
        if doc_length == 0:
            return 0
        return term_freq / doc_length

    def _compute_idf(self, term: str) -> float:
        """计算逆文档频率"""
        doc_freq = len(self.index.get(term, []))
        if doc_freq == 0:
            return 0
        return math.log(self.doc_count / doc_freq)

    def _tokenize(self, text: str) -> List[str]:
        """分词（支持中英文）"""
        text = text.lower()

        # 尝试使用 jieba 进行中文分词
        try:
            import jieba
            tokens = list(jieba.cut(text))
        except ImportError:
            # 简单的分词：按空格和标点分割
            tokens = re.findall(r'[\w\u4e00-\u9fff]+', text)

        # 过滤停用词和短词
        stop_words = {'的', '了', '是', '在', '和', '有', '不', '这', '我',
                      '你', '他', '她', '它', '们', '都', '也', '还', '就',
                      'the', 'a', 'an', 'is', 'are', 'was', 'were', 'in',
                      'on', 'at', 'to', 'for', 'of', 'with', 'and', 'or'}

        return [t for t in tokens if len(t) > 1 and t not in stop_words]

    def save(self, filepath: str):
        """保存索引到文件"""
        import json
        data = {
            'documents': {k: v.__dict__ for k, v in self.documents.items()},
            'index': {
                term: [p.__dict__ for p in postings]
                for term, postings in self.index.items()
            },
            'doc_count': self.doc_count,
        }
        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False)

    @classmethod
    def load(cls, filepath: str) -> 'InvertedIndex':
        """从文件加载索引"""
        import json
        with open(filepath, 'r', encoding='utf-8') as f:
            data = json.load(f)

        idx = cls()
        idx.doc_count = data['doc_count']

        for doc_id_str, doc_data in data['documents'].items():
            idx.documents[int(doc_id_str)] = DocumentInfo(**doc_data)

        for term, postings in data['index'].items():
            idx.index[term] = [PostingEntry(**p) for p in postings]

        return idx
```

### 第 3 步：查询解析器

```python
# query/parser.py
from dataclasses import dataclass
from typing import List, Optional
from enum import Enum, auto

class QueryType(Enum):
    KEYWORD = auto()    # 普通关键词搜索
    PHRASE = auto()     # 短语搜索 "exact phrase"
    AND = auto()        # AND 操作
    OR = auto()         # OR 操作
    NOT = auto()        # NOT 操作

@dataclass
class QueryNode:
    """查询树节点"""
    query_type: QueryType
    term: str = ""
    children: List['QueryNode'] = None

    def __post_init__(self):
        if self.children is None:
            self.children = []

class QueryParser:
    """查询解析器"""

    def parse(self, query_string: str) -> QueryNode:
        """解析查询字符串"""
        tokens = self._tokenize_query(query_string)

        if not tokens:
            return QueryNode(QueryType.KEYWORD, "")

        # 检查是否包含布尔操作符
        if 'AND' in tokens:
            return self._parse_and(tokens)
        elif 'OR' in tokens:
            return self._parse_or(tokens)
        elif 'NOT' in tokens:
            return self._parse_not(tokens)

        # 短语搜索
        if query_string.startswith('"') and query_string.endswith('"'):
            return QueryNode(QueryType.PHRASE, query_string.strip('"'))

        # 普通关键词搜索（多词视为 AND）
        if len(tokens) > 1:
            children = [QueryNode(QueryType.KEYWORD, t) for t in tokens]
            return QueryNode(QueryType.AND, children=children)

        return QueryNode(QueryType.KEYWORD, tokens[0])

    def _tokenize_query(self, query: str) -> List[str]:
        tokens = []
        current = []
        in_quotes = False

        for ch in query:
            if ch == '"':
                in_quotes = not in_quotes
                current.append(ch)
            elif ch == ' ' and not in_quotes:
                if current:
                    tokens.append(''.join(current))
                    current = []
            else:
                current.append(ch)

        if current:
            tokens.append(''.join(current))

        return tokens

    def _parse_and(self, tokens: List[str]) -> QueryNode:
        parts = [t for t in tokens if t != 'AND']
        children = [QueryNode(QueryType.KEYWORD, t) for t in parts]
        return QueryNode(QueryType.AND, children=children)

    def _parse_or(self, tokens: List[str]) -> QueryNode:
        parts = [t for t in tokens if t != 'OR']
        children = [QueryNode(QueryType.KEYWORD, t) for t in parts]
        return QueryNode(QueryType.OR, children=children)

    def _parse_not(self, tokens: List[str]) -> QueryNode:
        parts = [t for t in tokens if t != 'NOT']
        children = [QueryNode(QueryType.KEYWORD, t) for t in parts]
        return QueryNode(QueryType.NOT, children=children)
```

### 第 4 步：搜索引擎主类

```python
# engine/search_engine.py
from index.inverted_index import InvertedIndex
from crawler.spider import WebCrawler
from query.parser import QueryParser, QueryType

class SearchEngine:
    """搜索引擎"""

    def __init__(self):
        self.index = InvertedIndex()
        self.crawler = WebCrawler(max_pages=50, delay=0.5)
        self.parser = QueryParser()

    def crawl(self, seed_urls: list[str]):
        """爬取网页并建立索引"""
        pages = self.crawler.crawl(seed_urls)

        for page in pages:
            doc_id = self.index.add_document(page.url, page.title, page.content)

        print(f"\n索引完成: {len(pages)} 个文档, "
              f"{len(self.index.index)} 个词条")

    def search(self, query_string: str, top_k: int = 10) -> list:
        """搜索"""
        query_tree = self.parser.parse(query_string)
        return self._execute_query(query_tree, top_k)

    def _execute_query(self, query_node, top_k: int) -> list:
        """执行查询树"""
        if query_node.query_type == QueryType.KEYWORD:
            return self.index.search(query_node.term, top_k)

        elif query_node.query_type == QueryType.PHRASE:
            # 短语搜索：需要验证词的位置连续性
            results = self.index.search(query_node.term, top_k * 2)
            # 简化处理：返回匹配的结果
            return results[:top_k]

        elif query_node.query_type == QueryType.AND:
            # AND：所有子查询的结果交集
            if not query_node.children:
                return []
            all_results = [self._execute_query(child, top_k * 3)
                          for child in query_node.children]

            # 取交集并合并分数
            doc_scores = {}
            for result_set in all_results:
                for doc, score in result_set:
                    if doc.doc_id not in doc_scores:
                        doc_scores[doc.doc_id] = (doc, 0)
                    doc_scores[doc.doc_id] = (
                        doc, doc_scores[doc.doc_id][1] + score
                    )

            # 只保留在所有结果集中都出现的文档
            min_sets = len(all_results)
            final = [(doc, score) for doc, score in doc_scores.values()
                    if score > 0]
            final.sort(key=lambda x: x[1], reverse=True)
            return final[:top_k]

        elif query_node.query_type == QueryType.OR:
            all_results = []
            for child in query_node.children:
                all_results.extend(self._execute_query(child, top_k))
            # 去重并合并分数
            seen = {}
            for doc, score in all_results:
                if doc.doc_id not in seen or score > seen[doc.doc_id][1]:
                    seen[doc.doc_id] = (doc, score)
            result = sorted(seen.values(), key=lambda x: x[1], reverse=True)
            return result[:top_k]

        return []
```

```python
# web/app.py
from flask import Flask, render_template_string, request
from engine.search_engine import SearchEngine

app = Flask(__name__)
engine = SearchEngine()

SEARCH_TEMPLATE = """
<!DOCTYPE html>
<html>
<head>
    <title>MiniSearch - 搜索引擎</title>
    <style>
        body { font-family: Arial, sans-serif; max-width: 800px; margin: 50px auto; }
        .search-box { text-align: center; margin-bottom: 30px; }
        input[type="text"] { width: 500px; padding: 12px; font-size: 16px; border: 2px solid #ddd; border-radius: 24px; }
        button { padding: 12px 24px; font-size: 16px; background: #4285f4; color: white; border: none; border-radius: 24px; cursor: pointer; }
        .result { margin-bottom: 20px; }
        .result-url { color: #006621; font-size: 14px; }
        .result-title { color: #1a0dab; font-size: 18px; text-decoration: none; }
        .result-snippet { color: #545454; font-size: 14px; }
        .result-score { color: #999; font-size: 12px; }
    </style>
</head>
<body>
    <div class="search-box">
        <h1>🔍 MiniSearch</h1>
        <form method="GET" action="/search">
            <input type="text" name="q" value="{{ query }}" placeholder="输入搜索关键词...">
            <button type="submit">搜索</button>
        </form>
    </div>
    {% if results %}
    <p>找到 {{ results|length }} 个结果 (用时 {{ time }}ms)</p>
    {% for doc, score in results %}
    <div class="result">
        <a class="result-title" href="{{ doc.url }}">{{ doc.title }}</a>
        <div class="result-url">{{ doc.url }}</div>
        <div class="result-snippet">{{ doc.title }}</div>
        <span class="result-score">相关度: {{ "%.4f"|format(score) }}</span>
    </div>
    {% endfor %}
    {% endif %}
</body>
</html>
"""

@app.route('/')
def index():
    return render_template_string(SEARCH_TEMPLATE, query="", results=None)

@app.route('/search')
def search():
    query = request.args.get('q', '')
    import time as t
    start = t.time()
    results = engine.search(query) if query else []
    elapsed = round((t.time() - start) * 1000, 2)

    return render_template_string(SEARCH_TEMPLATE,
                                  query=query, results=results, time=elapsed)

if __name__ == '__main__':
    # 可选：先爬取一些页面
    # engine.crawl(["https://example.com"])
    app.run(port=8888, debug=True)
```
