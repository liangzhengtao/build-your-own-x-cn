## 📖 核心概念解释

### 1. 存储引擎架构

```
┌────────────────────────────────────────┐
│              SQL 层                     │
│  解析器 → 验证器 → 优化器 → 执行器     │
├────────────────────────────────────────┤
│            存储引擎层                   │
│  表管理器 → 页管理器 → 缓冲池管理器     │
├────────────────────────────────────────┤
│            文件系统层                   │
│         数据文件 (.db)                  │
└────────────────────────────────────────┘
```

### 2. 页（Page）的概念

数据库以"页"为单位读写磁盘，通常 4KB 一个页：

```
┌─────────────────────────────────────┐
│              数据页 (4KB)            │
├─────────────────────────────────────┤
│ 页头 (24 bytes)                     │
│  ├ 页ID (8 bytes)                   │
│  ├ 页类型 (2 bytes)                 │
│  ├ 记录数 (2 bytes)                 │
│  ├ 空闲空间偏移 (4 bytes)           │
│  └ 父页指针 (8 bytes)               │
├─────────────────────────────────────┤
│ 槽目录 (Slot Directory)             │
│  ├ Slot 0: offset, length           │
│  ├ Slot 1: offset, length           │
│  └ ...                              │
├─────────────────────────────────────┤
│ 空闲空间                            │
├─────────────────────────────────────┤
│ 记录数据 (从页尾向前增长)            │
│  └ Record N ... Record 1            │
└─────────────────────────────────────┘
```

### 3. B+ 树索引

B+ 树是数据库索引的标准数据结构：

```
            [30 | 60]                    ← 内部节点（只存键）
           /     |     \
    [10|20] [40|50] [70|80|90]           ← 内部节点
    / | \   / | \    /  |  \
  叶子节点  叶子节点  叶子节点             ← 叶子节点（存键+值，链表连接）
  ←→ ←→ ←→ ←→ ←→ ←→
```

**B+ 树的优势**：
- 所有数据都在叶子节点，范围查询高效
- 树高度低（3-4 层可索引百万条记录）
- 插入删除自动平衡

### 4. 查询处理流程

```
SQL 文本
  │
  ▼
词法分析器 (Tokenizer)
  │
  ▼
语法分析器 (Parser) → AST（抽象语法树）
  │
  ▼
语义分析器 (Validator) → 验证表名、列名
  │
  ▼
查询优化器 (Optimizer) → 选择最优执行计划
  │
  ▼
执行引擎 (Executor) → 调用存储引擎
  │
  ▼
结果集 (Result Set)
```

### 第 1 步：页式存储引擎

```python
# storage/page.py
import struct
from typing import Optional

PAGE_SIZE = 4096  # 4KB
PAGE_HEADER_SIZE = 24

# 页类型常量
PAGE_TYPE_INTERNAL = 1
PAGE_TYPE_LEAF = 2
PAGE_TYPE_DATA = 3

class Page:
    """数据库页的基本单元"""

    def __init__(self, page_id: int = 0, page_type: int = PAGE_TYPE_DATA):
        self.page_id = page_id
        self.page_type = page_type
        self.num_records = 0
        self.free_offset = PAGE_HEADER_SIZE
        self.data = bytearray(PAGE_SIZE)
        self.is_dirty = False

    def serialize(self) -> bytes:
        """将页序列化为字节流"""
        struct.pack_into('>Q', self.data, 0, self.page_id)
        struct.pack_into('>H', self.data, 8, self.page_type)
        struct.pack_into('>H', self.data, 10, self.num_records)
        struct.pack_into('>I', self.data, 12, self.free_offset)
        return bytes(self.data)

    @classmethod
    def deserialize(cls, raw_data: bytes) -> 'Page':
        """从字节流反序列化页"""
        page = cls()
        page.data = bytearray(raw_data)
        page.page_id = struct.unpack_from('>Q', raw_data, 0)[0]
        page.page_type = struct.unpack_from('>H', raw_data, 8)[0]
        page.num_records = struct.unpack_from('>H', raw_data, 10)[0]
        page.free_offset = struct.unpack_from('>I', raw_data, 12)[0]
        return page

    def insert_record(self, record: bytes) -> int:
        """插入一条记录，返回槽号"""
        record_len = len(record)
        if self.free_offset + record_len > PAGE_SIZE - (self.num_records + 1) * 4:
            raise OverflowError("页已满，无法插入更多记录")

        # 写入记录数据
        offset = self.free_offset
        self.data[offset:offset + record_len] = record

        # 更新槽目录（从页尾向前）
        slot_id = self.num_records
        slot_offset = PAGE_SIZE - (slot_id + 1) * 4
        struct.pack_into('>HH', self.data, slot_offset, offset, record_len)

        self.num_records += 1
        self.free_offset += record_len
        self.is_dirty = True

        return slot_id

    def get_record(self, slot_id: int) -> Optional[bytes]:
        """根据槽号读取记录"""
        if slot_id >= self.num_records:
            return None

        slot_offset = PAGE_SIZE - (slot_id + 1) * 4
        offset, length = struct.unpack_from('>HH', self.data, slot_offset)
        return bytes(self.data[offset:offset + length])
```

### 第 2 步：缓冲池管理器

```python
# storage/buffer_pool.py
from collections import OrderedDict
from .page import Page, PAGE_SIZE

class BufferPool:
    """LRU 缓冲池 - 管理内存中的数据页"""

    def __init__(self, pool_size: int = 128, db_file: str = "minidb.db"):
        self.pool_size = pool_size
        self.db_file = db_file
        self.pages: OrderedDict[int, Page] = OrderedDict()
        self.file = open(db_file, 'a+b')

    def get_page(self, page_id: int) -> Page:
        """获取指定页，如果不在内存中则从磁盘加载"""
        if page_id in self.pages:
            # 缓存命中，移到最近使用
            self.pages.move_to_end(page_id)
            return self.pages[page_id]

        # 缓存未命中，从磁盘加载
        if len(self.pages) >= self.pool_size:
            self._evict_page()

        page = self._read_from_disk(page_id)
        self.pages[page_id] = page
        return page

    def new_page(self) -> Page:
        """分配一个新页"""
        if len(self.pages) >= self.pool_size:
            self._evict_page()

        # 新页 ID 为当前最大 ID + 1
        page_id = max(self.pages.keys(), default=-1) + 1
        page = Page(page_id=page_id)
        self.pages[page_id] = page
        return page

    def flush_page(self, page_id: int):
        """将指定页写回磁盘"""
        if page_id in self.pages and self.pages[page_id].is_dirty:
            self._write_to_disk(self.pages[page_id])
            self.pages[page_id].is_dirty = False

    def flush_all(self):
        """将所有脏页写回磁盘"""
        for page_id in list(self.pages.keys()):
            self.flush_page(page_id)

    def _evict_page(self):
        """使用 LRU 策略淘汰一个页"""
        evicted_id, evicted_page = self.pages.popitem(last=False)
        if evicted_page.is_dirty:
            self._write_to_disk(evicted_page)

    def _read_from_disk(self, page_id: int) -> Page:
        """从磁盘读取页"""
        offset = page_id * PAGE_SIZE
        self.file.seek(offset)
        raw_data = self.file.read(PAGE_SIZE)

        if len(raw_data) < PAGE_SIZE:
            return Page(page_id=page_id)

        return Page.deserialize(raw_data)

    def _write_to_disk(self, page: Page):
        """将页写入磁盘"""
        offset = page.page_id * PAGE_SIZE
        self.file.seek(offset)
        self.file.write(page.serialize())
        self.file.flush()

    def close(self):
        """关闭缓冲池"""
        self.flush_all()
        self.file.close()
```

### 第 3 步：B+ 树索引

```python
# index/btree.py
import struct
from storage.page import Page, PAGE_SIZE

class BPlusTree:
    """B+ 树索引实现"""

    MAX_KEYS = 128  # 每个节点的最大键数

    def __init__(self, buffer_pool, root_page_id: int = None):
        self.bp = buffer_pool
        self.root_page_id = root_page_id

    def search(self, key: int) -> int:
        """搜索键对应的值（记录ID），O(log n)"""
        if self.root_page_id is None:
            return -1

        page = self.bp.get_page(self.root_page_id)

        while page.page_type != 2:  # 不是叶子节点
            keys, children = self._read_internal(page)
            child_idx = self._find_child_index(keys, key)
            page = self.bp.get_page(children[child_idx])

        # 在叶子节点中查找
        keys, values = self._read_leaf(page)
        for i, k in enumerate(keys):
            if k == key:
                return values[i]

        return -1  # 未找到

    def insert(self, key: int, value: int):
        """插入键值对"""
        if self.root_page_id is None:
            root = self.bp.new_page()
            root.page_type = 2  # 叶子节点
            self._write_leaf(root, [key], [value])
            self.root_page_id = root.page_id
            return

        # 递归插入（简化版：直接在叶子节点插入）
        path = []
        page = self.bp.get_page(self.root_page_id)

        while page.page_type != 2:
            keys, children = self._read_internal(page)
            child_idx = self._find_child_index(keys, key)
            path.append((page, child_idx))
            page = self.bp.get_page(children[child_idx])

        keys, values = self._read_leaf(page)
        insert_pos = 0
        while insert_pos < len(keys) and keys[insert_pos] < key:
            insert_pos += 1

        keys.insert(insert_pos, key)
        values.insert(insert_pos, value)

        if len(keys) <= self.MAX_KEYS:
            self._write_leaf(page, keys, values)
        else:
            self._split_leaf(page, keys, values, path)

    def _find_child_index(self, keys: list, key: int) -> int:
        """在内部节点中找到子节点索引"""
        for i, k in enumerate(keys):
            if key < k:
                return i
        return len(keys)

    def _read_internal(self, page: Page):
        """读取内部节点的键和子指针"""
        num_keys = struct.unpack_from('>H', page.data, 24)[0]
        keys = []
        children = []

        for i in range(num_keys):
            key = struct.unpack_from('>Q', page.data, 26 + i * 8)[0]
            keys.append(key)

        for i in range(num_keys + 1):
            child = struct.unpack_from('>Q', page.data,
                26 + self.MAX_KEYS * 8 + i * 8)[0]
            children.append(child)

        return keys, children

    def _read_leaf(self, page: Page):
        """读取叶子节点的键和值"""
        num_keys = struct.unpack_from('>H', page.data, 24)[0]
        keys = []
        values = []

        for i in range(num_keys):
            key = struct.unpack_from('>Q', page.data, 26 + i * 8)[0]
            value = struct.unpack_from('>Q',
                page.data, 26 + self.MAX_KEYS * 8 + i * 8)[0]
            keys.append(key)
            values.append(value)

        return keys, values

    def _write_leaf(self, page: Page, keys: list, values: list):
        """写入叶子节点"""
        page.page_type = 2
        struct.pack_into('>H', page.data, 24, len(keys))

        for i, key in enumerate(keys):
            struct.pack_into('>Q', page.data, 26 + i * 8, key)

        for i, value in enumerate(values):
            struct.pack_into('>Q', page.data,
                26 + self.MAX_KEYS * 8 + i * 8, value)

        page.is_dirty = True

    def _split_leaf(self, page, keys, values, path):
        """叶子节点分裂"""
        mid = len(keys) // 2

        # 左半部分
        self._write_leaf(page, keys[:mid], values[:mid])

        # 右半部分（新页）
        new_page = self.bp.new_page()
        new_page.page_type = 2
        self._write_leaf(new_page, keys[mid:], values[mid:])

        # 向上传播分裂
        self._propagate_split(path, keys[mid], new_page.page_id)

    def _propagate_split(self, path, promoted_key, right_child_id):
```

### 第 4 步：SQL 词法分析器

```python
# sql/tokenizer.py
from enum import Enum, auto
from dataclasses import dataclass
from typing import List

class TokenType(Enum):
    # 关键字
    SELECT = auto()
    FROM = auto()
    WHERE = auto()
    INSERT = auto()
    INTO = auto()
    VALUES = auto()
    CREATE = auto()
    TABLE = auto()
    UPDATE = auto()
    SET = auto()
    DELETE = auto()
    AND = auto()
    OR = auto()
    NOT = auto()
    NULL = auto()
    PRIMARY = auto()
    KEY = auto()
    INT = auto()
    VARCHAR = auto()

    # 运算符
    EQUALS = auto()      # =
    NOT_EQUALS = auto()   # <>
    LESS = auto()         # <
    GREATER = auto()      # >
    LESS_EQ = auto()      # <=
    GREATER_EQ = auto()   # >=
    PLUS = auto()         # +
    MINUS = auto()        # -
    STAR = auto()         # *
    COMMA = auto()        # ,
    SEMICOLON = auto()    # ;
    LPAREN = auto()       # (
    RPAREN = auto()       # )

    # 字面量
    INTEGER = auto()
    STRING = auto()
    IDENTIFIER = auto()

    EOF = auto()

KEYWORDS = {
    'SELECT': TokenType.SELECT, 'FROM': TokenType.FROM,
    'WHERE': TokenType.WHERE, 'INSERT': TokenType.INSERT,
    'INTO': TokenType.INTO, 'VALUES': TokenType.VALUES,
    'CREATE': TokenType.CREATE, 'TABLE': TokenType.TABLE,
    'UPDATE': TokenType.UPDATE, 'SET': TokenType.SET,
    'DELETE': TokenType.DELETE, 'AND': TokenType.AND,
    'OR': TokenType.OR, 'NOT': TokenType.NOT,
    'NULL': TokenType.NULL, 'PRIMARY': TokenType.PRIMARY,
    'KEY': TokenType.KEY, 'INT': TokenType.INT,
    'VARCHAR': TokenType.VARCHAR,
}

@dataclass
class Token:
    type: TokenType
    value: str
    line: int
    column: int

class Tokenizer:
    """SQL 词法分析器"""

    def __init__(self, sql: str):
        self.sql = sql
        self.pos = 0
        self.line = 1
        self.column = 1
        self.tokens: List[Token] = []

    def tokenize(self) -> List[Token]:
        """将 SQL 文本转换为 Token 列表"""
        while self.pos < len(self.sql):
            self._skip_whitespace()

            if self.pos >= len(self.sql):
                break

            ch = self.sql[self.pos]

            # 标识符或关键字
            if ch.isalpha() or ch == '_':
                self._read_identifier()
            # 数字
            elif ch.isdigit():
                self._read_number()
            # 字符串
            elif ch == "'":
                self._read_string()
            # 运算符和标点
            else:
                self._read_operator()

        self.tokens.append(Token(TokenType.EOF, '', self.line, self.column))
        return self.tokens

    def _skip_whitespace(self):
        while self.pos < len(self.sql) and self.sql[self.pos] in ' \t\n\r':
            if self.sql[self.pos] == '\n':
                self.line += 1
                self.column = 1
            else:
                self.column += 1
            self.pos += 1

    def _read_identifier(self):
        start = self.pos
        col = self.column
        while self.pos < len(self.sql) and (self.sql[self.pos].isalnum() or self.sql[self.pos] == '_'):
            self.pos += 1
            self.column += 1

        value = self.sql[start:self.pos].upper()
        token_type = KEYWORDS.get(value, TokenType.IDENTIFIER)
        self.tokens.append(Token(token_type, value, self.line, col))

    def _read_number(self):
        start = self.pos
        col = self.column
        while self.pos < len(self.sql) and self.sql[self.pos].isdigit():
            self.pos += 1
            self.column += 1
        self.tokens.append(Token(TokenType.INTEGER, self.sql[start:self.pos], self.line, col))

    def _read_string(self):
        col = self.column
        self.pos += 1  # 跳过开始的 '
        self.column += 1
        start = self.pos

        while self.pos < len(self.sql) and self.sql[self.pos] != "'":
            self.pos += 1
            self.column += 1

        value = self.sql[start:self.pos]
        self.pos += 1   # 跳过结束的 '
        self.column += 1
        self.tokens.append(Token(TokenType.STRING, value, self.line, col))

    def _read_operator(self):
        ch = self.sql[self.pos]
        col = self.column
        ops = {
            '=': TokenType.EQUALS, '*': TokenType.STAR,
            ',': TokenType.COMMA, ';': TokenType.SEMICOLON,
            '(': TokenType.LPAREN, ')': TokenType.RPAREN,
            '+': TokenType.PLUS, '-': TokenType.MINUS,
        }

        if ch in ops:
            self.tokens.append(Token(ops[ch], ch, self.line, col))
            self.pos += 1
            self.column += 1
        elif ch == '<':
            if self.pos + 1 < len(self.sql) and self.sql[self.pos + 1] == '>':
                self.tokens.append(Token(TokenType.NOT_EQUALS, '<>', self.line, col))
                self.pos += 2
            elif self.pos + 1 < len(self.sql) and self.sql[self.pos + 1] == '=':
                self.tokens.append(Token(TokenType.LESS_EQ, '<=', self.line, col))
                self.pos += 2
            else:
                self.tokens.append(Token(TokenType.LESS, '<', self.line, col))
                self.pos += 1
            self.column += 2
        elif ch == '>':
            if self.pos + 1 < len(self.sql) and self.sql[self.pos + 1] == '=':
                self.tokens.append(Token(TokenType.GREATER_EQ, '>=', self.line, col))
                self.pos += 2
            else:
                self.tokens.append(Token(TokenType.GREATER, '>', self.line, col))
                self.pos += 1
            self.column += 2
        else:
            raise SyntaxError(f"无法识别的字符 '{ch}' 在第 {self.line} 行")
```

### 第 5 步：SQL 解析器

```python
# sql/parser.py
from dataclasses import dataclass
from typing import List, Optional, Any
from .tokenizer import Token, TokenType

@dataclass
class Column:
    name: str
    col_type: str
    is_primary_key: bool = False

@dataclass
class CreateTableStmt:
    table_name: str
    columns: List[Column]

@dataclass
class InsertStmt:
    table_name: str
    columns: Optional[List[str]]
    values: List[List[Any]]

@dataclass
class SelectStmt:
    columns: List[str]         # ['*'] 表示全部
    table_name: str
    where_clause: Optional[Any] = None

@dataclass
class BinaryExpr:
    left: Any
    operator: str
    right: Any

class Parser:
    """递归下降 SQL 解析器"""

    def __init__(self, tokens: List[Token]):
        self.tokens = tokens
        self.pos = 0

    def parse(self):
        """解析 SQL 语句"""
        stmt = self._parse_statement()
        self._expect(TokenType.SEMICOLON)
        return stmt

    def _current(self) -> Token:
        return self.tokens[self.pos]

    def _advance(self) -> Token:
        token = self.tokens[self.pos]
        self.pos += 1
        return token

    def _expect(self, token_type: TokenType) -> Token:
        token = self._current()
        if token.type != token_type:
            raise SyntaxError(
                f"期望 {token_type.name}，但得到 {token.type.name} "
                f"'{token.value}' 在第 {token.line} 行"
            )
        return self._advance()

    def _parse_statement(self):
        token = self._current()
        if token.type == TokenType.CREATE:
            return self._parse_create_table()
        elif token.type == TokenType.SELECT:
            return self._parse_select()
        elif token.type == TokenType.INSERT:
            return self._parse_insert()
        else:
            raise SyntaxError(f"不支持的语句: {token.value}")

    def _parse_create_table(self) -> CreateTableStmt:
        self._expect(TokenType.CREATE)
        self._expect(TokenType.TABLE)
        table_name = self._expect(TokenType.IDENTIFIER).value
        self._expect(TokenType.LPAREN)

        columns = []
        columns.append(self._parse_column_def())
        while self._current().type == TokenType.COMMA:
            self._advance()
            columns.append(self._parse_column_def())

        self._expect(TokenType.RPAREN)
        return CreateTableStmt(table_name, columns)

    def _parse_column_def(self) -> Column:
        name = self._expect(TokenType.IDENTIFIER).value
        col_type = self._expect(TokenType.IDENTIFIER).value

        is_pk = False
        if self._current().type == TokenType.PRIMARY:
            self._advance()
            self._expect(TokenType.KEY)
            is_pk = True

        return Column(name, col_type, is_pk)

    def _parse_select(self) -> SelectStmt:
        self._expect(TokenType.SELECT)
        columns = self._parse_column_list()
        self._expect(TokenType.FROM)
        table_name = self._expect(TokenType.IDENTIFIER).value

        where_clause = None
        if self._current().type == TokenType.WHERE:
            self._advance()
            where_clause = self._parse_expression()

        return SelectStmt(columns, table_name, where_clause)

    def _parse_insert(self) -> InsertStmt:
        self._expect(TokenType.INSERT)
        self._expect(TokenType.INTO)
        table_name = self._expect(TokenType.IDENTIFIER).value

        columns = None
        if self._current().type == TokenType.LPAREN:
            columns = self._parse_column_list()

        self._expect(TokenType.VALUES)
        values = [self._parse_value_list()]

        while self._current().type == TokenType.COMMA:
            self._advance()
            values.append(self._parse_value_list())

        return InsertStmt(table_name, columns, values)

    def _parse_column_list(self) -> List[str]:
        if self._current().type == TokenType.STAR:
            self._advance()
            return ['*']

        columns = [self._expect(TokenType.IDENTIFIER).value]
        while self._current().type == TokenType.COMMA:
            self._advance()
            columns.append(self._expect(TokenType.IDENTIFIER).value)
        return columns

    def _parse_value_list(self) -> List[Any]:
        self._expect(TokenType.LPAREN)
        values = [self._parse_literal()]
        while self._current().type == TokenType.COMMA:
            self._advance()
            values.append(self._parse_literal())
        self._expect(TokenType.RPAREN)
        return values

    def _parse_literal(self):
        token = self._current()
        if token.type == TokenType.INTEGER:
            self._advance()
            return int(token.value)
        elif token.type == TokenType.STRING:
            self._advance()
            return token.value
        elif token.type == TokenType.NULL:
            self._advance()
            return None
        else:
            raise SyntaxError(f"期望字面量，但得到 {token.value}")

    def _parse_expression(self):
        left = self._parse_comparison()

        while self._current().type in (TokenType.AND, TokenType.OR):
            op = self._advance().value
            right = self._parse_comparison()
            left = BinaryExpr(left, op, right)

        return left

    def _parse_comparison(self):
        left = self._parse_primary()

        comp_ops = {
            TokenType.EQUALS: '=', TokenType.NOT_EQUALS: '<>',
            TokenType.LESS: '<', TokenType.GREATER: '>',
            TokenType.LESS_EQ: '<=', TokenType.GREATER_EQ: '>=',
        }

        if self._current().type in comp_ops:
            op = comp_ops[self._advance().type]
            right = self._parse_primary()
            return BinaryExpr(left, op, right)

        return left

    def _parse_primary(self):
        token = self._current()
        if token.type == TokenType.IDENTIFIER:
            self._advance()
            return token.value
        elif token.type == TokenType.INTEGER:
            self._advance()
            return int(token.value)
        elif token.type == TokenType.STRING:
            self._advance()
            return token.value
        elif token.type == TokenType.LPAREN:
            self._advance()
            expr = self._parse_expression()
            self._expect(TokenType.RPAREN)
            return expr
        else:
            raise SyntaxError(f"无法解析: {token.value}")
```

### 第 6 步：执行引擎

```python
# engine/executor.py
from storage.buffer_pool import BufferPool
from index.btree import BPlusTree

class Table:
    """表的内存表示"""
    def __init__(self, name: str, columns: list, page_id: int):
        self.name = name
        self.columns = columns      # [(name, type, is_pk), ...]
        self.page_id = page_id
        self.indexes: dict = {}     # {col_name: BPlusTree}

class MiniDB:
    """MiniDB 数据库引擎"""

    def __init__(self, db_path: str = "minidb.db"):
        self.buffer_pool = BufferPool(db_file=db_path)
        self.tables: dict[str, Table] = {}
        self._load_catalog()

    def execute(self, sql: str) -> list:
        """执行 SQL 语句"""
        from sql.tokenizer import Tokenizer
        from sql.parser import Parser, CreateTableStmt, SelectStmt, InsertStmt

        tokens = Tokenizer(sql).tokenize()
        stmt = Parser(tokens).parse()

        if isinstance(stmt, CreateTableStmt):
            return self._exec_create_table(stmt)
        elif isinstance(stmt, InsertStmt):
            return self._exec_insert(stmt)
        elif isinstance(stmt, SelectStmt):
            return self._exec_select(stmt)
        else:
            raise ValueError(f"不支持的语句类型: {type(stmt)}")

    def _exec_create_table(self, stmt: CreateTableStmt) -> list:
        """执行 CREATE TABLE"""
        page = self.buffer_pool.new_page()
        columns = [(col.name, col.col_type, col.is_primary_key) for col in stmt.columns]

        table = Table(stmt.table_name, columns, page.page_id)
        self.tables[stmt.table_name] = table

        # 为主键创建 B+ 树索引
        for col in stmt.columns:
            if col.is_primary_key:
                idx_page = self.buffer_pool.new_page()
                table.indexes[col.name] = BPlusTree(self.buffer_pool, idx_page.page_id)

        return [{"status": "OK", "message": f"表 '{stmt.table_name}' 创建成功"}]

    def _exec_insert(self, stmt: InsertStmt) -> list:
        """执行 INSERT INTO"""
        if stmt.table_name not in self.tables:
            raise ValueError(f"表 '{stmt.table_name}' 不存在")

        table = self.tables[stmt.table_name]
        results = []

        for values in stmt.values:
            page = self.buffer_pool.get_page(table.page_id)

            # 序列化记录
            record = self._serialize_record(table, dict(zip(
                [c[0] for c in table.columns], values
            )))

            slot_id = page.insert_record(record)
            results.append({
                "status": "OK",
                "message": f"插入成功，slot_id={slot_id}"
            })

            # 更新索引
            for i, col in enumerate(table.columns):
                if col[0] in table.indexes:
                    table.indexes[col[0]].insert(values[i], slot_id)

        return results

    def _exec_select(self, stmt: SelectStmt) -> list:
        """执行 SELECT"""
        if stmt.table_name not in self.tables:
            raise ValueError(f"表 '{stmt.table_name}' 不存在")

        table = self.tables[stmt.table_name]
        page = self.buffer_pool.get_page(table.page_id)
        results = []

        for slot_id in range(page.num_records):
            record = page.get_record(slot_id)
            if record:
                row = self._deserialize_record(table, record)

                # 应用 WHERE 条件
                if stmt.where_clause and not self._eval_where(row, stmt.where_clause):
                    continue

                # 选择列
                if stmt.columns == ['*']:
                    results.append(row)
                else:
                    results.append({col: row[col] for col in stmt.columns if col in row})

        return results

    def _serialize_record(self, table: Table, row: dict) -> bytes:
        """将行序列化为字节"""
        import struct
        parts = []
        for col_name, col_type, _ in table.columns:
            value = row.get(col_name)
            if col_type.upper() == 'INT':
                parts.append(struct.pack('>i', value if value else 0))
            elif col_type.upper().startswith('VARCHAR'):
                s = str(value or '')
                parts.append(struct.pack('>H', len(s)) + s.encode('utf-8'))
        return b''.join(parts)

    def _deserialize_record(self, table: Table, data: bytes) -> dict:
        """将字节反序列化为行"""
        import struct
        row = {}
        offset = 0
        for col_name, col_type, _ in table.columns:
            if col_type.upper() == 'INT':
                row[col_name] = struct.unpack_from('>i', data, offset)[0]
                offset += 4
            elif col_type.upper().startswith('VARCHAR'):
                length = struct.unpack_from('>H', data, offset)[0]
                offset += 2
                row[col_name] = data[offset:offset + length].decode('utf-8')
                offset += length
        return row

    def _eval_where(self, row: dict, expr) -> bool:
        """求值 WHERE 条件"""
        if hasattr(expr, 'operator'):
            left = row.get(expr.left, expr.left) if isinstance(expr.left, str) else expr.left
            right = row.get(expr.right, expr.right) if isinstance(expr.right, str) else expr.right

            ops = {
                '=': lambda a, b: a == b,
                '<>': lambda a, b: a != b,
                '<': lambda a, b: a < b,
                '>': lambda a, b: a > b,
                '<=': lambda a, b: a <= b,
                '>=': lambda a, b: a >= b,
            }
            return ops[expr.operator](left, right)
        return True

    def _load_catalog(self):
        """加载系统目录（简化版）"""
        pass

    def close(self):
        """关闭数据库"""
        self.buffer_pool.close()
```

```python
# main.py
"""MiniDB - 从零构建的关系型数据库"""
from engine.executor import MiniDB

def main():
    print("╔══════════════════════════════════════╗")
    print("║     MiniDB v0.1 - 教学数据库引擎     ║")
    print("║  输入 SQL 语句，或 'quit' 退出       ║")
    print("╚══════════════════════════════════════╝")
    print()

    db = MiniDB("test.db")

    while True:
        try:
            sql = input("minidb> ").strip()
            if not sql:
                continue
            if sql.lower() in ('quit', 'exit', 'q'):
                break
            if not sql.endswith(';'):
                # 多行输入
                while not sql.endswith(';'):
                    sql += ' ' + input("     > ").strip()

            results = db.execute(sql)
            for row in results:
                print(row)

        except Exception as e:
            print(f"错误: {e}")

    db.close()
    print("再见！")

if __name__ == "__main__":
    main()
```
